> ⚠️ 平台仅支持提取文档类文件（TXT/MD/PDF/DOCX）。若用户上传图片版简历，需提示"请上传 PDF 或 Word 格式"。

### 3.3 A3 · 代码节点（简历结构化）

**输入变量**：`text`　**输出变量**：`profile`

```python
import re
import json


def main(text):
    t = text or ""
    t = re.sub(r"[ \t\u3000]+", " ", t)
    lines = [l.strip() for l in t.split("\n") if l.strip()]

    # 1. 联系方式
    phone = re.search(r"1[3-9]\d{9}", t)
    email = re.search(r"[\w.\-]+@[\w\-]+\.[a-z]{2,}", t, re.I)

    # 2. 技能关键词（内置词典，可按专业扩充）
    SKILL_DICT = [
        "python", "java", "sql", "c++", "javascript", "git", "linux",
        "aspen", "autocad", "origin", "matlab", "hplc", "gc", "icp",
        "excel", "vue", "react", "plc", "solidworks", "ansys",
        "化工原理", "物料衡算", "实验设计", "正交实验", "数据分析",
        "文献调研", "工艺流程", "hazop",
    ]
    low = t.lower()
    skills = [s for s in SKILL_DICT if s.lower() in low]

    # 3. 量化结果（含数字+单位的句子）
    quant = re.findall(
        r"[^。\n；;]{0,40}\d+(?:\.\d+)?\s*(?:%|％|万|人|条|次|小时|分钟|天|元|kg|g|ml)[^。\n；;]{0,20}",
        t,
    )

    # 4. 动词强度统计
    weak = sum(len(re.findall(w, t)) for w in ["负责", "参与", "协助", "帮忙"])
    strong = sum(len(re.findall(w, t)) for w in ["设计", "实现", "搭建", "优化", "主导", "独立"])

    # 5. 模块识别
    SECTIONS = ["教育背景", "教育经历", "专业技能", "技能", "项目经历", "项目经验",
                "实习经历", "实践经历", "校园经历", "荣誉", "获奖", "自我评价"]
    found = []
    for ln in lines:
        for s in SECTIONS:
            if s in ln and len(ln) <= 12 and s not in found:
                found.append(s)

    profile = {
        "char_count": len(t),
        "has_phone": bool(phone),
        "has_email": bool(email),
        "skills": skills,
        "skill_count": len(skills),
        "quantified_items": quant[:8],
        "quantified_count": len(quant),
        "weak_verbs": weak,
        "strong_verbs": strong,
        "sections_found": found,
        "raw_text": t[:6000],
    }
    return {"profile": json.dumps(profile, ensure_ascii=False)}
```

> 💡 **答辩亮点**：代码节点做了 **5 类结构化抽取**（联系方式 / 技能词典匹配 / 量化结果 / 动词强度 / 模块识别），
> 为诊断提供**客观数据**，而不是把整篇简历直接丢给大模型——这是"工程化"和"套壳"的分界线。

```python
def main(cond_json, limit=10):
    import json
    c = json.loads(cond_json or "{}")
    where = ["1=1"]
    if c.get("city"):
        where.append(f"city LIKE '%{c['city']}%'")
    if c.get("job_type"):
        where.append(f"job_type = '{c['job_type']}'")
    if c.get("major"):
        where.append(f"major_req LIKE '%{c['major']}%'")
    if c.get("industry"):
        where.append(f"industry LIKE '%{c['industry']}%'")
    for s in (c.get("skills") or []):
        where.append(f"(must_skills LIKE '%{s}%' OR plus_skills LIKE '%{s}%')")
    sql = ("SELECT id, job_title, company_type, industry, city, job_type, degree, "
           "major_req, must_skills, plus_skills, duties, salary_range FROM jobs "
           "WHERE " + " AND ".join(where) + f" LIMIT {limit}")
    return {"sql": sql}
```


### 4.4 B3 · 代码节点（规则匹配度打分）

**输入变量**：`jobs`（B2 查询结果）、`user_major`、`user_skills`
**输出变量**：`scored_jobs`

```python
import json


def main(jobs, user_major, user_skills):
    if isinstance(jobs, str):
        try:
            jobs = json.loads(jobs)
        except Exception:
            jobs = []

    # 用户技能标准化
    if isinstance(user_skills, str):
        u_skills = [s.strip().lower() for s in user_skills.replace("，", ",").split(",") if s.strip()]
    else:
        u_skills = [str(s).strip().lower() for s in (user_skills or [])]

    u_major = (user_major or "").strip()

    out = []
    for j in jobs:
        must = [s.strip().lower() for s in str(j.get("must_skills", "")).split(";") if s.strip()]
        plus = [s.strip().lower() for s in str(j.get("plus_skills", "")).split(";") if s.strip()]

        # 技能命中率（35%）
        if must:
            hit = sum(1 for m in must if any(m in u or u in m for u in u_skills))
            skill_score = hit / len(must) * 100
        else:
            skill_score = 60

        # 专业相关度（40%）
        majors = [m.strip() for m in str(j.get("major_req", "")).split(";") if m.strip()]
        if not majors or any("不限" in m for m in majors):
            major_score = 75
        elif any(u_major and (u_major in m or m in u_major) for m in majors):
            major_score = 100
        else:
            major_score = 55

        # 加分技能（10%）
        plus_hit = sum(1 for p in plus if any(p in u or u in p for u in u_skills))
        plus_score = min(100, plus_hit / len(plus) * 100) if plus else 60

        # 地域与经历相关度（15%）交由下游大模型补充，这里给基准 75
        score = round(major_score * 0.40 + skill_score * 0.35 + plus_score * 0.10 + 75 * 0.15)

        item = dict(j)
        item["rule_score"] = score
        item["missing_skills"] = [m for m in must if not any(m in u or u in m for u in u_skills)]
        out.append(item)

    out.sort(key=lambda x: -x["rule_score"])
    return {"scored_jobs": json.dumps(out[:10], ensure_ascii=False)}
```
